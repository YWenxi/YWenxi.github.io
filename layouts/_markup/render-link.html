{{- $text := strings.TrimSpace .Text -}} {{- $dest := .Destination -}} {{- $bare := or (eq $text "") (eq $text $dest) -}} {{- if and $bare (not .Title) (or (hasPrefix $dest "https://") (hasPrefix $dest "http://")) -}} {{/* Notion-style mention: favicon + page title for bare links. */}} {{- $host := (urls.Parse $dest).Host -}} {{- $title := "" -}} {{- $r := try (resources.GetRemote $dest (dict "timeout" "5s")) -}} {{- if eq $r.Err nil -}} {{- with $r.Value -}} {{- $matches := findRESubmatch "(?is)]*>(.*?)" .Content 1 -}} {{- if gt (len $matches) 0 -}} {{- $title = index (index $matches 0) 1 | htmlUnescape | strings.TrimSpace -}} {{- end -}} {{- end -}} {{- end -}} {{- if eq $title "" }}{{ $title = $host }}{{ end -}} {{ $title }} {{- else -}} {{ .Text | safeHTML }} {{- end -}}
